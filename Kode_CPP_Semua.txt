#include <bits/stdc++.h>
using namespace std;

int a[100005], pre[100005], suf[100005];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];

    // pre[i] = fpb dari a[1..i], suf[i] = fpb dari a[i..n]
    pre[0] = 0;
    for (int i = 1; i <= n; i++) pre[i] = __gcd(pre[i - 1], a[i]);
    suf[n + 1] = 0;
    for (int i = n; i >= 1; i--) suf[i] = __gcd(suf[i + 1], a[i]);

    int idxBest = 1;
    int gcdBest = -1;
    // buang a[i] sampai a[i+k-1], sisanya pre[i-1] dan suf[i+k]
    for (int i = 1; i + k - 1 <= n; i++) {
        int g = __gcd(pre[i - 1], suf[i + k]);
        if (g > gcdBest) {
            gcdBest = g;
            idxBest = i;
        }
    }

    cout << idxBest << " " << gcdBest << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

long long A[1005][1005];
long long E[1005][1005];     // hasil kali 1 baris tanpa kolom j
long long hasil[1005][1005];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n, m;
    long long k;
    cin >> n >> m >> k;

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cin >> A[i][j];
            A[i][j] = A[i][j] % k;
        }
    }

    // K bukan prima jadi ga bisa dibagi, pakai prefix sama suffix aja
    // langkah 1: tiap baris, kali semua kecuali kolom j
    for (int i = 1; i <= n; i++) {
        vector<long long> kiri(m + 2), kanan(m + 2);
        kiri[0] = 1 % k;
        kanan[m + 1] = 1 % k;
        for (int j = 1; j <= m; j++) kiri[j] = kiri[j - 1] * A[i][j] % k;
        for (int j = m; j >= 1; j--) kanan[j] = kanan[j + 1] * A[i][j] % k;
        for (int j = 1; j <= m; j++) E[i][j] = kiri[j - 1] * kanan[j + 1] % k;
    }

    // langkah 2: tiap kolom, kali E semua baris kecuali baris i
    for (int j = 1; j <= m; j++) {
        vector<long long> atas(n + 2), bawah(n + 2);
        atas[0] = 1 % k;
        bawah[n + 1] = 1 % k;
        for (int i = 1; i <= n; i++) atas[i] = atas[i - 1] * E[i][j] % k;
        for (int i = n; i >= 1; i--) bawah[i] = bawah[i + 1] * E[i][j] % k;
        for (int i = 1; i <= n; i++) hasil[i][j] = atas[i - 1] * bawah[i + 1] % k;
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            cout << hasil[i][j];
            if (j < m) cout << " ";
        }
        cout << "\n";
    }
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int n, m;
string grid[1005];
int kuat[1005][1005];

// terima satu garis (daftar koordinat), cari palindrom terpanjang tiap pusat pakai manacher
void proses(vector<pair<int, int> > &garis) {
    int len = garis.size();
    string s = "";
    for (int i = 0; i < len; i++) {
        s += grid[garis[i].first][garis[i].second];
    }

    vector<int> d(len);
    int l = 0, r = -1;
    for (int i = 0; i < len; i++) {
        int k;
        if (i > r) k = 1;
        else k = min(d[l + r - i], r - i + 1);
        while (i - k >= 0 && i + k < len && s[i - k] == s[i + k]) k++;
        d[i] = k;
        if (i + k - 1 > r) {
            l = i - k + 1;
            r = i + k - 1;
        }
    }

    for (int i = 0; i < len; i++) {
        int panjang = 2 * d[i] - 1;
        int x = garis[i].first, y = garis[i].second;
        kuat[x][y] = min(kuat[x][y], panjang);
    }
}

void jalan(int si, int sj, int di, int dj) {
    vector<pair<int, int> > garis;
    int i = si, j = sj;
    while (i >= 0 && i < n && j >= 0 && j < m) {
        garis.push_back(make_pair(i, j));
        i += di;
        j += dj;
    }
    proses(garis);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> grid[i];

    for (int i = 0; i < n; i++)
        for (int j = 0; j < m; j++)
            kuat[i][j] = 1000000000;

    // horizontal
    for (int i = 0; i < n; i++) jalan(i, 0, 0, 1);
    // vertikal
    for (int j = 0; j < m; j++) jalan(0, j, 1, 0);
    // diagonal turun (\)
    for (int j = 0; j < m; j++) jalan(0, j, 1, 1);
    for (int i = 1; i < n; i++) jalan(i, 0, 1, 1);
    // diagonal naik (/)
    for (int j = 0; j < m; j++) jalan(0, j, 1, -1);
    for (int i = 1; i < n; i++) jalan(i, m - 1, 1, -1);

    int terbesar = 0, banyak = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (kuat[i][j] > terbesar) {
                terbesar = kuat[i][j];
                banyak = 1;
            } else if (kuat[i][j] == terbesar) {
                banyak++;
            }
        }
    }

    cout << banyak << " " << terbesar << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int n, q;
vector<int> tree[800005];   // segment tree, tiap node nyimpen daftar shield
long long hp[200005];
long long habis[200005];    // waktu terakhir shield masih aktif
int jumlahShield = 0;

// masukin shield id ke semua node yang pas nutup interval a..b
void pasang(int node, int kiri, int kanan, int a, int b, int id) {
    if (a <= kiri && kanan <= b) {
        tree[node].push_back(id);
        return;
    }
    int tengah = (kiri + kanan) / 2;
    if (a <= tengah) pasang(node * 2, kiri, tengah, a, b, id);
    if (b > tengah) pasang(node * 2 + 1, tengah + 1, kanan, a, b, id);
}

// cari shield paling baru yang masih hidup di wilayah a
int cari(int a, int waktu) {
    int node = 1, kiri = 1, kanan = n;
    int terbaik = 0;
    while (true) {
        // buang shield yang udah mati dari belakang
        while (!tree[node].empty()) {
            int id = tree[node].back();
            if (hp[id] <= 0 || habis[id] < waktu) {
                tree[node].pop_back();
            } else {
                break;
            }
        }
        if (!tree[node].empty()) {
            terbaik = max(terbaik, tree[node].back());
        }
        if (kiri == kanan) break;
        int tengah = (kiri + kanan) / 2;
        if (a <= tengah) {
            node = node * 2;
            kanan = tengah;
        } else {
            node = node * 2 + 1;
            kiri = tengah + 1;
        }
    }
    return terbaik;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> q;
    for (int t = 1; t <= q; t++) {
        int tipe;
        cin >> tipe;
        if (tipe == 1) {
            int a, b;
            long long d, h;
            cin >> a >> b >> d >> h;
            jumlahShield++;
            hp[jumlahShield] = h;
            habis[jumlahShield] = t + d - 1;
            pasang(1, 1, n, a, b, jumlahShield);
        } else {
            int a;
            long long k;
            cin >> a >> k;
            int id = cari(a, t);
            if (id == 0) {
                cout << "EVAKUASI SEKARANG!\n";
            } else {
                hp[id] -= k;
                cout << "Shield Elmax pelindung kota!\n";
            }
        }
    }
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    int n, m;
    cin >> n >> m;

    vector<long long> semua;
    long long normal = 0;
    for (int i = 0; i < n; i++) {
        long long maks = 0;
        for (int j = 0; j < m; j++) {
            long long x;
            cin >> x;
            semua.push_back(x);
            maks = max(maks, x);
        }
        normal += maks;   // normal: ambil terbesar tiap ruangan
    }

    // reversal: bebas tukar, jadi ambil n hadiah terbesar dari semuanya
    sort(semua.begin(), semua.end(), greater<long long>());
    long long rev = 0;
    for (int i = 0; i < n; i++) rev += semua[i];

    cout << normal << " " << rev << endl;
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int n;
pair<int, int> a[300005];      // (L, R) udah diurutin
int sp[20][300005];            // sparse table buat min R
int lg[300005];
int m1[300005], idx1[300005], m2[300005];   // R terbesar & terbesar kedua dari awal
int kompDari[300005];
int kompAwal[300005], kompAkhir[300005];
long long nilaiKomp[300005];
vector<int> potong[300005];

int minR(int l, int r) {
    int k = lg[r - l + 1];
    return min(sp[k][l], sp[k][r - (1 << k) + 1]);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    cin >> n;
    for (int i = 1; i <= n; i++) {
        cin >> a[i].first >> a[i].second;
    }
    sort(a + 1, a + n + 1);

    // sparse table
    lg[1] = 0;
    for (int i = 2; i <= n; i++) lg[i] = lg[i / 2] + 1;
    for (int i = 1; i <= n; i++) sp[0][i] = a[i].second;
    for (int k = 1; (1 << k) <= n; k++) {
        for (int i = 1; i + (1 << k) - 1 <= n; i++) {
            sp[k][i] = min(sp[k - 1][i], sp[k - 1][i + (1 << (k - 1))]);
        }
    }

    // bagi jadi rombongan (komponen)
    int jumlahKomp = 0;
    int maxR = -1;
    for (int i = 1; i <= n; i++) {
        if (i == 1 || a[i].first > maxR) {
            jumlahKomp++;
            kompAwal[jumlahKomp] = i;
        }
        kompDari[i] = jumlahKomp;
        kompAkhir[jumlahKomp] = i;
        maxR = max(maxR, a[i].second);
    }
    for (int c = 1; c <= jumlahKomp; c++) {
        int s = kompAwal[c], e = kompAkhir[c];
        long long v = (long long)minR(s, e) - a[e].first + 1;
        if (v < 0) v = 0;
        nilaiKomp[c] = v;
    }

    // R terbesar dan terbesar kedua dari prefix
    int b1 = -1, b2 = -1, bi = 0;
    for (int i = 1; i <= n; i++) {
        int r = a[i].second;
        if (r > b1) {
            b2 = b1;
            b1 = r;
            bi = i;
        } else if (r > b2) {
            b2 = r;
        }
        m1[i] = b1;
        idx1[i] = bi;
        m2[i] = b2;
    }

    // titik potong: kalau satu orang dibuang rombongan kepisah di q
    for (int q = 1; q < n; q++) {
        int L = a[q + 1].first;
        if (m1[q] >= L && m2[q] < L) {
            potong[idx1[q]].push_back(q);
        }
    }

    // nilai rombongan terbaik dan kedua terbaik
    long long best1 = 0, best2 = 0;
    int best1komp = 0;
    for (int c = 1; c <= jumlahKomp; c++) {
        if (nilaiKomp[c] > best1) {
            best2 = best1;
            best1 = nilaiKomp[c];
            best1komp = c;
        } else if (nilaiKomp[c] > best2) {
            best2 = nilaiKomp[c];
        }
    }

    long long jawab = 0;
    for (int x = 1; x <= n; x++) {
        int c = kompDari[x];
        int s = kompAwal[c], e = kompAkhir[c];
        long long cand;
        if (best1komp != c) cand = best1;
        else cand = best2;

        int e1;
        if (potong[x].empty()) e1 = e;
        else e1 = potong[x][0];

        // grup pertama (x dibuang)
        int mn = INT_MAX;
        if (x - 1 >= s) mn = min(mn, minR(s, x - 1));
        if (x + 1 <= e1) mn = min(mn, minR(x + 1, e1));
        if (mn != INT_MAX) {
            int mx;
            if (e1 == x) mx = a[x - 1].first;
            else mx = a[e1].first;
            cand = max(cand, (long long)mn - mx + 1);
        }

        // grup-grup lainnya
        for (int t = 0; t < (int)potong[x].size(); t++) {
            int kiri = potong[x][t] + 1;
            int kanan;
            if (t + 1 < (int)potong[x].size()) kanan = potong[x][t + 1];
            else kanan = e;
            cand = max(cand, (long long)minR(kiri, kanan) - a[kanan].first + 1);
        }

        jawab = max(jawab, cand);
    }

    if (jawab <= 0) cout << -1 << endl;
    else cout << jawab << endl;
    return 0;
}
